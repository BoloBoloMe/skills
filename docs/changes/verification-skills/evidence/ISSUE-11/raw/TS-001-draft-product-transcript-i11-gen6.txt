import json
import os
from pathlib import Path


def data_dir() -> Path:
    return Path(os.environ.get("NOTES_DATA_DIR", ".notes-data"))


def _file() -> Path:
    directory = data_dir()
    directory.mkdir(parents=True, exist_ok=True)
    return directory / "notes.json"


def load() -> list[dict]:
    path = _file()
    if not path.exists():
        return []
    return json.loads(path.read_text(encoding="utf-8"))


def save(items: list[dict]) -> None:
    _file().write_text(json.dumps(items, ensure_ascii=False, indent=2), encoding="utf-8")


def add(text: str) -> dict:
    items = load()
    item = {"id": len(items) + 1, "text": text}
    items.append(item)
    save(items)
    return item

import argparse
import html
import json
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
from urllib.parse import parse_qs

from . import store


class Handler(BaseHTTPRequestHandler):
    def _send(self, status: int, content_type: str, body: bytes) -> None:
        self.send_response(status)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self) -> None:
        if self.path.startswith("/notes"):
            body = json.dumps(store.load(), ensure_ascii=False).encode("utf-8")
            self._send(200, "application/json; charset=utf-8", body)
            return
        rows = "".join(f"<li>{html.escape(item['text'])}</li>" for item in store.load())
        page = f"<!doctype html><html><head><meta charset=\"utf-8\"><title>笔记</title></head><body><h1>笔记</h1><ul>{rows}</ul></body></html>"
        self._send(200, "text/html; charset=utf-8", page.encode("utf-8"))

    def do_POST(self) -> None:
        length = int(self.headers.get("Content-Length") or 0)
        raw = self.rfile.read(length).decode("utf-8")
        if self.headers.get("Content-Type", "").startswith("application/json"):
            text = json.loads(raw)["text"]
        else:
            text = parse_qs(raw).get("text", [""])[0]
        item = store.add(text)
        body = json.dumps(item, ensure_ascii=False).encode("utf-8")
        self._send(201, "application/json; charset=utf-8", body)

    def log_message(self, fmt: str, *args) -> None:
        print(fmt % args, flush=True)
