"""逐场景运行 PRODUCT.md 中的命令, 记录 stdout/stderr/退出码并做断言."""
import json
import os
import shutil
import subprocess
import sys
import time
import urllib.request
from pathlib import Path

ROOT = Path("/tmp/orchestra-02/red-coverage")
EVD = ROOT / "docs/changes/tool-cli/evidence-v1-bb25738"
ENV = {**os.environ, "PYTHONDONTWRITEBYTECODE": "1"}


def run_case(cid, argv, checks):
    d = EVD / cid
    d.mkdir(parents=True, exist_ok=True)
    cmd = "python3 " + " ".join(argv)
    r = subprocess.run(["python3", *argv], cwd=ROOT, capture_output=True, text=True, env=ENV)
    (d / "cmd.txt").write_text(cmd + "\n", encoding="utf-8")
    (d / "argv.json").write_text(json.dumps(["python3", *argv], ensure_ascii=False) + "\n", encoding="utf-8")
    (d / "stdout.txt").write_text(r.stdout, encoding="utf-8")
    (d / "stderr.txt").write_text(r.stderr, encoding="utf-8")
    (d / "exitcode.txt").write_text(str(r.returncode) + "\n", encoding="utf-8")
    results = []
    for label, fn in checks:
        results.append((label, bool(fn(r))))
    passed = all(ok for _, ok in results)
    (d / "checks.txt").write_text(
        "".join(f"{'PASS' if ok else 'FAIL'}  {label}\n" for label, ok in results), encoding="utf-8"
    )
    return {
        "id": cid, "cmd": cmd, "exit": r.returncode,
        "stdout": r.stdout, "stderr": r.stderr,
        "checks": results, "verdict": "通过" if passed else "失败",
    }


def start_stub():
    p = subprocess.Popen(
        ["python3", "/tmp/acceptance-bb25738/health_stub.py"],
        stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, env=ENV,
    )
    for _ in range(50):
        try:
            urllib.request.urlopen("http://127.0.0.1:18080/health", timeout=1).read()
            return p
        except Exception:
            time.sleep(0.1)
    p.kill()
    raise SystemExit("health stub failed to start")

    lines.append(f"exit: {c['exit']}")
    lines.append("stdout: " + json.dumps(c["stdout"], ensure_ascii=False))
    lines.append("stderr: " + json.dumps(c["stderr"], ensure_ascii=False))
